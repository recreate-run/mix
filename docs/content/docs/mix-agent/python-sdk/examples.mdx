Here are some common usage examples for the Mix Python SDK.

## Setting Up the Client

```python
from mix_python_sdk import Mix

# Synchronous client
client = Mix(server_url="http://localhost:8088")

# Using a context manager (recommended)
with Mix(server_url="http://localhost:8088") as client:
    # Your code here
    pass
```

## Asynchronous Client

```python
import asyncio
from mix_python_sdk import Mix

async def main():
    async with Mix(server_url="http://localhost:8088") as client:
        # Your async code here
        pass

# Run the async function
asyncio.run(main())
```

## Working with Sessions

### Creating a New Session

```python
def create_new_session():
    try:
        with Mix(server_url="http://localhost:8088") as client:
            session = client.sessions.create(title="My New Session")
            print(f"Created session: {session.id}")
            return session
    except Exception as e:
        print(f"Failed to create session: {e}")
        raise
```

### Listing All Sessions

```python
def list_all_sessions():
    try:
        with Mix(server_url="http://localhost:8088") as client:
            sessions = client.sessions.list()
            print(f"Found {len(sessions)} sessions")
            return sessions
    except Exception as e:
        print(f"Failed to list sessions: {e}")
        raise
```

### Getting a Specific Session

```python
def get_session_by_id(session_id):
    try:
        with Mix(server_url="http://localhost:8088") as client:
            session = client.sessions.get(id=session_id)
            print(f"Retrieved session: {session.title}")
            return session
    except Exception as e:
        print(f"Failed to get session {session_id}: {e}")
        raise
```

### Deleting a Session

```python
def delete_session(session_id):
    try:
        with Mix(server_url="http://localhost:8088") as client:
            client.sessions.delete(id=session_id)
            print(f"Session {session_id} deleted successfully")
    except Exception as e:
        print(f"Failed to delete session {session_id}: {e}")
        raise
```

## Selecting Provider and Model

```python
def select_provider_and_model():
    try:
        with Mix(server_url="http://localhost:8088") as client:
            # Get available providers and models
            providers = client.preferences.get_available_providers()
            print(f"Available providers: {list(providers.keys())}")
            
            # Set preferred provider and model
            client.preferences.update_preferences(
                preferred_provider="anthropic",
                main_agent_model="claude-4-sonnet"
            )
            
            print("Provider and model selected successfully")
    except Exception as e:
        print(f"Failed to select provider and model: {e}")
        raise
```

## Async Examples

### Creating a Session and Sending a Message

```python
import asyncio
from mix_python_sdk import Mix

async def create_session_and_chat():
    async with Mix(server_url="http://localhost:8088") as client:
        # Create a new session
        session = await client.sessions.create(title="Async Chat Session")
        print(f"Created session: {session.id}")
        
        # Send a message
        message = await client.messages.send(
            session_id=session.id,
            content="Hello, can you help me with Python programming?"
        )
        print(f"Sent message: {message.id}")
        
        # Wait a moment for the assistant to respond
        await asyncio.sleep(2)
        
        # Get all messages
        messages = await client.messages.get_session(session_id=session.id)
        for msg in messages:
            print(f"{msg.role}: {msg.content[:50]}...")

# Run the async function
asyncio.run(create_session_and_chat())
```

## Integration with FastAPI

For web applications using FastAPI, you can integrate with the Mix Python SDK:

```python
from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import List, Optional
from mix_python_sdk import Mix

app = FastAPI()
client = Mix(server_url="http://localhost:8088")

class SessionCreate(BaseModel):
    title: str

class MessageCreate(BaseModel):
    content: str

@app.post("/sessions/")
async def create_session(session_create: SessionCreate):
    try:
        session = await client.sessions.create(title=session_create.title)
        return session
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/sessions/")
async def list_sessions():
    try:
        return await client.sessions.list()
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/sessions/{session_id}")
async def get_session(session_id: str):
    try:
        return await client.sessions.get(id=session_id)
    except Exception as e:
        raise HTTPException(status_code=404, detail="Session not found")

@app.post("/sessions/{session_id}/messages")
async def send_message(session_id: str, message_create: MessageCreate):
    try:
        return await client.messages.send(
            session_id=session_id,
            content=message_create.content
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/sessions/{session_id}/messages")
async def get_messages(session_id: str):
    try:
        return await client.messages.get_session(session_id=session_id)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

## Complete Example Application

This example demonstrates a simple command-line chat application using the Mix SDK:

```python
import argparse
import sys
from mix_python_sdk import Mix

def main():
    parser = argparse.ArgumentParser(description="Mix SDK Chat Client")
    parser.add_argument("--server", default="http://localhost:8088", help="Mix server URL")
    parser.add_argument("--provider", default="anthropic", help="AI provider to use")
    parser.add_argument("--model", default="claude-4-sonnet", help="AI model to use")
    args = parser.parse_args()
    
    print(f"Connecting to Mix server at {args.server}...")
    
    try:
        with Mix(server_url=args.server) as client:
            # Configure provider and model
            client.preferences.update_preferences(
                preferred_provider=args.provider,
                main_agent_model=args.model
            )
            print(f"Using {args.provider} with model {args.model}")
            
            # Create a new session
            session = client.sessions.create(title="CLI Chat Session")
            print(f"Created new session: {session.title} (ID: {session.id})")
            
            print("\nEnter your messages (type 'exit' to quit):")
            
            while True:
                user_input = input("\nYou: ").strip()
                if user_input.lower() == "exit":
                    break
                
                # Send user message
                client.messages.send(session_id=session.id, content=user_input)
                
                # Get all messages to display the response
                messages = client.messages.get_session(session_id=session.id)
                
                # Print the latest assistant message
                assistant_messages = [m for m in messages if m.role == "assistant"]
                if assistant_messages:
                    latest = assistant_messages[-1]
                    print(f"\nAssistant: {latest.content}")
            
            print(f"\nEnding session. Session ID: {session.id}")
    
    except Exception as e:
        print(f"Error: {e}")
        return 1
    
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

These examples demonstrate common patterns for using the Mix Python SDK in both synchronous and asynchronous contexts.

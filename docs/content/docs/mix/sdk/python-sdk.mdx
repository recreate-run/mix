## Quickstart

```python
from mix_python_sdk import Mix
import os
from dotenv import load_dotenv
from utils import stream_message


def main():
    load_dotenv()
    api_key = os.getenv("OPENROUTER_API_KEY")
    if not api_key:
        raise ValueError("OPENROUTER_API_KEY not found in environment variables")

    user_msg = "Say hi"

    with Mix(server_url=os.getenv("MIX_SERVER_URL")) as mix:
        mix.system.get_health()
        mix.authentication.store_api_key(api_key=api_key, provider="openrouter")

        # session creation
        session = mix.sessions.create(title="Streaming Demo")
        stream_message(mix, session.id, user_msg)


if __name__ == "__main__":
    main()
```

For more examples, see the [Mix Python cookbook](https://github.com/recreate-run/mix-cookbooks/tree/main/python_examples)

## Creating the client

### Synchronous Client

```python
## Setting Up the Client
from mix_python_sdk import Mix

# Synchronous client
client = Mix(server_url="http://localhost:8088")

# Using a context manager (recommended)
with Mix(server_url="http://localhost:8088") as client:
    # Your code here
    pass
```

### Asynchronous Client

```python
import asyncio
from mix_python_sdk import Mix

async def main():
    async with Mix(server_url="http://localhost:8088") as client:
        # Your async code here
        pass

# Run the async function
asyncio.run(main())
```

## Working with Sessions
### Creating a New Session

```python
def create_new_session():
    try:
        with Mix(server_url="http://localhost:8088") as client:
            session = client.sessions.create(title="My New Session")
            print(f"Created session: {session.id}")
            return session
    except Exception as e:
        print(f"Failed to create session: {e}")
        raise
```

### Listing All Sessions

```python
def list_all_sessions():
```

```python
def get_session_by_id(session_id):
```

## Selecting Provider and Model

```python
def select_provider_and_model():
    try:
        with Mix(server_url="http://localhost:8088") as client:
            # Get available providers and models
            providers = client.preferences.get_available_providers()
            print(f"Available providers: {list(providers.keys())}")
            
            # Set preferred provider and model
            client.preferences.update_preferences(
                preferred_provider="anthropic",
                main_agent_model="claude-4-sonnet"
            )
            
            print("Provider and model selected successfully")
    except Exception as e:
        print(f"Failed to select provider and model: {e}")
        raise
```

## Working with Messages

### Sending a Message

```python
def send_message(session_id, content):
    try:
        with Mix(server_url="http://localhost:8088") as client:
```

## Tool Models

### show_media Tool

```python
from mix_python_sdk.tool_models import MediaShowcaseParams, MediaOutput, MediaType

# MediaType enum
class MediaType(str, Enum):
    IMAGE = "image"
    VIDEO = "video"
    AUDIO = "audio"
    GSAP_ANIMATION = "gsap_animation"
    PDF = "pdf"
    CSV = "csv"

# MediaOutput class
class MediaOutput(BaseModel):
    path: Optional[str]              # Path or URL to media file
    type: MediaType                   # Type of media
    title: str                        # Title for the output
    description: Optional[str]        # Optional description
    config: Optional[Dict[str, Any]]  # Config for gsap_animation
    start_time: Optional[int]         # Start time (seconds) for video/audio
    duration: Optional[int]           # Duration (seconds) for video/audio

# MediaShowcaseParams class
class MediaShowcaseParams(BaseModel):
    outputs: List[MediaOutput]        # Array of media outputs
```

### ReadMedia Tool

```python
from mix_python_sdk.tool_models import (
    ReadMediaParams, ReadMediaResult, ReadMediaResponse, MediaAnalysisType
)

# MediaAnalysisType enum
class MediaAnalysisType(str, Enum):
    IMAGE = "image"
    AUDIO = "audio"
    VIDEO = "video"
    PDF = "pdf"

# ReadMediaParams class
class ReadMediaParams(BaseModel):
    file_path: str                    # Path or URL to file
    media_type: MediaAnalysisType     # Type of media
    prompt: str                       # Analysis prompt
    pdf_pages: Optional[str]          # Page selection (PDF only)
    video_interval: Optional[str]     # Time interval (video only)

# ReadMediaResult class
class ReadMediaResult(BaseModel):
    file_path: str                    # Path to analyzed file
    media_type: str                   # Type of media analyzed
    analysis: str                     # Analysis results
    error: Optional[str]              # Error message if failed

# ReadMediaResponse class
class ReadMediaResponse(BaseModel):
    results: List[ReadMediaResult]    # List of analysis results
    summary: str                      # Overall summary
```

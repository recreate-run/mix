Mix provides comprehensive file handling capabilities within sessions, allowing you to upload documents, images, data files, and more for AI analysis and processing.

## Overview

Files in Mix are:
- **Session-scoped** - Each session has its own file storage
- **Type-agnostic** - Support for any file type (images, PDFs, CSVs, etc.)
- **Persistent** - Files remain available throughout session lifetime
- **Accessible to AI** - AI can read, analyze, and process uploaded files

## Uploading Files

### Basic Upload

```python
from mix_python_sdk import Mix

with Mix(server_url="http://localhost:8088") as mix:
    session = mix.sessions.create(title="File Analysis")

    # Upload file
    with open("data.csv", "rb") as f:
        file_info = mix.files.upload_session_file(
            id=session.id,
            file={
                "file_name": "data.csv",
                "content": f,
                "content_type": "text/csv",
            },
        )

    print(f"Uploaded: {file_info.name}")
    print(f"Size: {file_info.size} bytes")
    print(f"URL: {file_info.url}")
```

### Upload Multiple Files

```python
import os
from pathlib import Path

def upload_directory(mix: Mix, session_id: str, directory: str):
    """Upload all files from a directory"""

    dir_path = Path(directory)
    uploaded_files = []

    for file_path in dir_path.glob("*"):
        if file_path.is_file():
            # Determine content type
            content_type = get_content_type(file_path.suffix)

            with open(file_path, "rb") as f:
                file_info = mix.files.upload_session_file(
                    id=session_id,
                    file={
                        "file_name": file_path.name,
                        "content": f,
                        "content_type": content_type,
                    },
                )
                uploaded_files.append(file_info)
                print(f"✅ Uploaded: {file_info.name}")

    return uploaded_files

def get_content_type(extension: str) -> str:
    """Map file extension to content type"""
    content_types = {
        ".csv": "text/csv",
        ".txt": "text/plain",
        ".json": "application/json",
        ".jpg": "image/jpeg",
        ".jpeg": "image/jpeg",
        ".png": "image/png",
        ".pdf": "application/pdf",
    }
    return content_types.get(extension.lower(), "application/octet-stream")

# Usage
uploaded = upload_directory(mix, session.id, "./documents")
print(f"Uploaded {len(uploaded)} files")
```

```python
def upload_with_progress(mix: Mix, session_id: str, file_path: str):
```

### Download All Files

```python
import os

def download_all_files(mix: Mix, session_id: str, output_dir: str):
    """Download all session files to directory"""

    os.makedirs(output_dir, exist_ok=True)
    files = mix.files.list_session_files(id=session_id)

    for file_info in files:
        if file_info.is_dir:
            continue

        # Download file
        response = mix.files.get_session_file(
            id=session_id,
            filename=file_info.name
        )

        # Save to disk
        output_path = os.path.join(output_dir, file_info.name)
        with open(output_path, "wb") as f:
            f.write(response.read())

        print(f"Downloaded: {file_info.name}")

# Usage
download_all_files(mix, session.id, "./downloads")
```

```python
def download_image_with_thumbnail(mix: Mix, session_id: str, filename: str):
```

```python
def delete_files_by_pattern(mix: Mix, session_id: str, pattern: str):
    """Delete files matching pattern"""
    files = mix.files.list_session_files(id=session_id)

    deleted = 0
    for file in files:
        if pattern in file.name:
            mix.files.delete_session_file(
                id=session_id,
                filename=file.name
            )
            print(f"Deleted: {file.name}")
            deleted += 1

    print(f"Deleted {deleted} files")

# Usage - delete all CSV files
delete_files_by_pattern(mix, session.id, ".csv")
```

## Using Files with AI

### Analyze Uploaded File

```python
from pathlib import Path

def analyze_file(mix: Mix, session_id: str, file_path: str, question: str):
    """Upload file and ask AI to analyze it"""

    # Upload file
    with open(file_path, "rb") as f:
        file_info = mix.files.upload_session_file(
            id=session_id,
            file={
                "file_name": Path(file_path).name,
                "content": f,
                "content_type": get_content_type(Path(file_path).suffix),
            },
        )

    print(f"Uploaded: {file_info.url}")

    # Ask AI to analyze
    prompt = f"Look at @{file_info.url} and {question}"

    response = mix.messages.send_message(
        id=session_id,
        request_body={"text": prompt}
    )

    return response.content

# Usage
analysis = analyze_file(
    mix,
    session.id,
    "sales_data.csv",
    "find the top 5 products by revenue"
)
print(analysis)
```

### Process Multiple Files

```python
def analyze_dataset(mix: Mix, session_id: str, files: list[str]):
    """Upload multiple files and analyze together"""

    # Upload all files
    file_urls = []
    for file_path in files:
        with open(file_path, "rb") as f:
            file_info = mix.files.upload_session_file(
                id=session_id,
                file={
                    "file_name": Path(file_path).name,
                    "content": f,
                    "content_type": get_content_type(Path(file_path).suffix),
                },
            )
            file_urls.append(file_info.url)

    # Reference all files in prompt
    file_refs = " and ".join([f"@{url}" for url in file_urls])
    prompt = f"Compare the data in {file_refs}. What are the key insights?"

    response = mix.messages.send_message(
        id=session_id,
        request_body={"text": prompt}
    )

    return response.content

# Usage
insights = analyze_dataset(
    mix,
    session.id,
    ["Q1_sales.csv", "Q2_sales.csv", "Q3_sales.csv"]
)
```

## Common Patterns

### Upload, Analyze, Download Results

```python
def full_analysis_workflow(mix: Mix, input_file: str, output_file: str):
    """Complete workflow: upload → analyze → download results"""

    with Mix(server_url=os.getenv("MIX_SERVER_URL")) as mix:
        session = mix.sessions.create(title="Analysis Workflow")

        try:
            # 1. Upload input file
            with open(input_file, "rb") as f:
                file_info = mix.files.upload_session_file(
                    id=session.id,
                    file={
                        "file_name": Path(input_file).name,
                        "content": f,
                        "content_type": "text/csv",
                    },
                )

            # 2. Analyze and generate visualization
            response = mix.messages.send_message(
                id=session.id,
                request_body={
                    "text": f"Analyze @{file_info.url} and create a visualization. Save it as analysis_chart.png"
                }
            )

            print(response.content)

            # 3. Download generated chart
            chart = mix.files.get_session_file(
                id=session.id,
                filename="analysis_chart.png"
            )

            with open(output_file, "wb") as f:
                f.write(chart.read())

            print(f"Chart saved to {output_file}")

        finally:
            mix.sessions.delete(id=session.id)

# Usage
full_analysis_workflow("data.csv", "chart.png")
```

### Batch File Processing

```python
import concurrent.futures

def process_files_parallel(mix: Mix, file_paths: list[str], processor_fn):
    """Process multiple files in parallel sessions"""

    def process_one(file_path):
        session = mix.sessions.create(title=f"Process {Path(file_path).name}")

        try:
            return processor_fn(mix, session.id, file_path)
        finally:
            mix.sessions.delete(id=session.id)

    with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
        results = list(executor.map(process_one, file_paths))

    return results

# Usage
def analyze_sentiment(mix, session_id, file_path):
    # Upload and analyze
    with open(file_path, "rb") as f:
        file_info = mix.files.upload_session_file(id=session_id, ...)

    response = mix.messages.send_message(
        id=session_id,
        request_body={"text": f"Analyze sentiment in @{file_info.url}"}
    )
    return response.content

results = process_files_parallel(mix, csv_files, analyze_sentiment)
```

## Best Practices

### 1. Clean Up Files

```python
# ✅ Good - Clean up when done
try:
    # Upload and process
    file_info = mix.files.upload_session_file(...)
    # Process file
finally:
    mix.files.delete_session_file(id=session.id, filename=file_info.name)
```

### 2. Validate File Sizes

```python
# ✅ Good - Check file size before upload
MAX_SIZE = 10 * 1024 * 1024  # 10 MB

file_size = os.path.getsize(file_path)
if file_size > MAX_SIZE:
    raise ValueError(f"File too large: {file_size:,} bytes (max {MAX_SIZE:,})")
```

### 3. Handle Upload Errors

```python
# ✅ Good - Graceful error handling
try:
    file_info = mix.files.upload_session_file(...)
except Exception as e:
    print(f"Upload failed: {e}")
    # Retry or fallback logic
```
